49def build_network(n_features, n_classes): 

50 """The REQ-TR-3 architecture: Dense 128, BatchNorm, Dropout 0.3, Dense 64, Dropout 0.2, 

51 Dense 32, then a sigmoid (binary) or softmax (multi-class) output.""" 

52 binary = n_classes == 2 

53 network = keras.Sequential( 

54 [ 

55 keras.Input(shape=(n_features,)), 

56 keras.layers.Dense(128, activation="relu"), 

57 keras.layers.BatchNormalization(), 

58 keras.layers.Dropout(0.3), 

59 keras.layers.Dense(64, activation="relu"), 

60 keras.layers.Dropout(0.2), 

61 keras.layers.Dense(32, activation="relu"), 

62 keras.layers.Dense( 

63 1 if binary else n_classes, activation="sigmoid" if binary else "softmax" 

64 ), 

65 ] 

66 ) 

67 network.compile( 

68 optimizer=keras.optimizers.Adam(learning_rate=LEARNING_RATE), 

69 loss="binary_crossentropy" if binary else "sparse_categorical_crossentropy", 

70 metrics=["accuracy"], 

71 ) 

72 return network

84def class_weights(labels): 

85 """Keras ``class_weight``: each class weighted so all classes count equally.""" 

86 labels = np.asarray(labels) 

87 weights = balanced_weights(labels) 

88 return {int(label): float(weights[labels == label][0]) for label in np.unique(labels)}

91class DNNModel: 

92 """A trained network with the scikit-learn ``predict`` / ``predict_proba`` interface.""" 

93 

94 def __init__(self, network, n_classes): 

95 self.network = network 

96 self.n_classes = n_classes 

97 

98 def predict_proba(self, features): 

99 """Class probabilities, shape ``(rows, n_classes)``.""" 

100 output = self.network.predict(np.asarray(features, dtype=np.float32), verbose=0) 

101 if self.n_classes == 2: 

102 positive = output.ravel() 

103 return np.column_stack([1 - positive, positive]) 

104 return output 

105 

106 def predict(self, features): 

107 """The most probable class for each row.""" 

108 return np.argmax(self.predict_proba(features), axis=1)
