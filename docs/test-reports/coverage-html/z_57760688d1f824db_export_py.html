1"""CSV export of the metrics table (SM-65). 

2 

3Implements REQ-RS-9 and SEC-2: any text cell that starts with =, +, - or @ (or a tab or 

4carriage return) is prefixed with an apostrophe, so a spreadsheet shows it as text instead 

5of running it as a formula. Model names are fixed, but error messages and the dataset name 

6can carry text from the uploaded file. 

7""" 

8 

9import csv 

10import io 

11 

12FORMULA_PREFIXES = ("=", "+", "-", "@", "\t", "\r") 

13COLUMNS = ( 

14 ("Model", "model"), 

15 ("Paradigm", "paradigm"), 

16 ("Status", "status"), 

17 ("Accuracy", "accuracy"), 

18 ("Precision", "precision"), 

19 ("Recall", "recall"), 

20 ("F1", "f1"), 

21 ("ROC-AUC", "roc_auc"), 

22 ("PR-AUC", "pr_auc"), 

23 ("Threshold", "threshold"), 

24 ("Train time (s)", "train_time"), 

25 ("Test time (s)", "test_time"), 

26 ("Test rows", "test_rows"), 

27 ("Error", "error"), 

28) 

29 

30 

31def escape_cell(value): 

32 """A CSV-safe cell: numbers unchanged, None empty, formula-like text neutralised.""" 

33 if value is None: 

34 return "" 

35 if isinstance(value, float): 

36 return round(value, 6) 

37 if isinstance(value, (int, bool)): 

38 return value 

39 text = str(value) 

40 return f"'{text}" if text.startswith(FORMULA_PREFIXES) else text 

41 

42 

43def results_csv(results): 

44 """The metrics table as CSV text, one row per model, finished models first.""" 

45 buffer = io.StringIO() 

46 writer = csv.writer(buffer) 

47 writer.writerow([label for label, _ in COLUMNS]) 

48 ordered = sorted(results, key=lambda result: result.get("status") != "done") 

49 for result in ordered: 

50 writer.writerow([escape_cell(result.get(key)) for _, key in COLUMNS]) 

51 return buffer.getvalue()
